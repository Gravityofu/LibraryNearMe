# 개선112 - 게시판 기능 1단계: 데이터베이스 구조 만들기

## 목표

도서관 홈페이지에 게시판 기능(공지, 열린 게시판, 자료를 신청합니다 등 10종류)을 만들기 위한 첫 단계입니다.
이번 단계에서는 화면(관리자 페이지, 홈페이지)은 아직 만들지 않고, 데이터를 저장할 **데이터베이스 구조**만 먼저 만듭니다.

이번에 새로 만드는 테이블은 4개입니다.

- **Board(게시판)**: "공지", "열린 게시판" 같은 게시판 종류 자체를 저장합니다. 게시판 이름, 목록을 제목으로 보여줄지 썸네일로 보여줄지, 회원/비회원 글쓰기·댓글 허용 여부 같은 설정을 담습니다.
- **Post(게시글)**: 각 게시판에 올라오는 글입니다. 회원이 쓴 글인지 비회원이 쓴 글인지 구분해서 저장하고, 비회원 글은 나중에 본인이 입력한 비밀번호로 수정/삭제할 수 있도록 비밀번호도 암호화해서 저장합니다.
- **Comment(댓글)**: 글에 달리는 댓글입니다. 게시글처럼 회원/비회원 구분과 비회원 비밀번호를 저장합니다.
- **MaterialRequest(자료신청 추가정보)**: "자료를 신청합니다" 게시판의 글에만 붙는 추가 정보입니다. 신청하는 자료 종류(도서/DVD/공구 등), 저자, 그리고 관리자가 바꿀 수 있는 처리 상태(신청 / 구입 진행 중 / 구입 완료 / 미구입)를 저장합니다.

게시판 10개(새로 들어온 자료, 컬렉션, 참고서비스, 스크랩, 오늘 마주친 한 구절, 공지, 소식, 열린 게시판, 자주 묻는 질문, 자료를 신청합니다)를 실제로 만드는 작업은 다음 단계(2단계, 관리자 설정 화면)에서 진행합니다. 이번 단계는 그 그릇(테이블)만 만드는 단계입니다.

```prisma
  reservations  Reservation[]  // ← 추가 (예약 기록)

// 사용자 (회원 / 관리자 / 슈퍼계정 모두 여기에 담습니다)
model User {
  id            Int        @id @default(autoincrement())

  libraryId     Int                          // 어느 도서관 소속인지
  library       Library    @relation(fields: [libraryId], references: [id])

  name          String                       // 이름 (필수)
  phone         String?                      // 휴대폰 번호. 중복 허용 (가입 화면에서 입력받음)
  phoneVerified Boolean    @default(false)    // 번호 인증 여부 (나중 문자 인증용)
  email         String?                      // 이메일 (선택)
  birthDate     DateTime?                     // 생년월일 (선택)
  address       String?                       // 주소 (도로명 주소, 선택)
  addressDetail String?                       // 상세 주소 (동/호수 등, 선택)
  postalCode    String?                       // 우편번호 (선택)

  memberNo      String?                       // 회원증 번호(사람이 읽는). 기존 "U012345" 그대로 or 신규 발급
  cardToken     String     @unique            // QR 보안 코드(기계용). 무작위 고유값

  memberTypeId  Int?                          // 회원 구분 (역할이 "회원"일 때만 사용)
  memberType    MemberType? @relation(fields: [memberTypeId], references: [id])

  role          Role       @default(MEMBER)   // 역할
  status        UserStatus @default(ACTIVE)   // 계정 상태

  provider      String?                       // 소셜 로그인 경로 (나중용, 지금 비움)
  providerId    String?                       // 소셜 계정 식별값 (나중용, 지금 비움)

  createdAt     DateTime   @default(now())

  loans         Loan[]     // ← 추가 (이 회원의 대출 기록)
  loanRestrictions LoanRestriction[] // ← 추가 (이 회원의 대출제한 내역)
  reservations  Reservation[]        // ← 추가 (이 회원의 예약 기록)
  posts         Post[]               // ← 추가 (이 회원이 쓴 게시글)
  comments      Comment[]            // ← 추가 (이 회원이 쓴 댓글)

  @@unique([libraryId, loginId])              // 같은 도서관 안에서 아이디 중복 금지
  @@unique([libraryId, memberNo])             // 같은 도서관 안에서 회원증번호 중복 금지
}

enum UserStatus {

// 자료 정보 (서지 레코드)
model Material {
  id          Int      @id @default(autoincrement())

  libraryId   Int
  library     Library  @relation(fields: [libraryId], references: [id])

  type        String   // 자료 종류 코드 (book, dvd, thesis, ...)
  title       String   // 서명 — MARC 245 ▼a
  creator     String?  // 저자 — MARC 100(있으면) / 없으면 245 ▼d
  publisher   String?  // 발행처 — MARC 260 ▼b
  pubYear     String?  // 발행년 — MARC 260 ▼c
  isbn        String?  // ISBN — MARC 020 ▼a
  classNumber String?  // 분류기호 — MARC 056 (책·DVD 자동추출, 비도서는 선택 입력)
  format      String?  // 형태(크기·파일형식) — MARC 300
  subject     String?  // 주제어 — MARC 653
  language    String?  // 언어 — MARC 041
  summary     String?  // 요약/설명 — MARC 500 ▼a
  coverUrl    String?  // 표지 이미지 주소
  onlineUrl   String?  // 온라인 접근 주소

  marc        Json?    // 구조화된 MARC(태그 목록) — 책·DVD 서지 원본(편집본)
  marcRaw     String?  // KOLIS-NET 원본 문자열 그대로 보관
  extra       Json?    // 종류별 특수 정보

  createdById Int?     // 이 자료를 등록한 사서
  createdBy   User?    @relation(fields: [createdById], references: [id])

  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  copies      Copy[]

  @@index([libraryId])
}

// 실물 개별 자료 (소장 부수) — 실물이 있는 모든 종류에 등록
model Copy {
  id             Int        @id @default(autoincrement())

  libraryId      Int
  library        Library    @relation(fields: [libraryId], references: [id])
  materialId     Int
  material       Material   @relation(fields: [materialId], references: [id])

  registrationNo String     // 등록번호(고유번호) — 도서관 내 고유
  callNumber     String?    // 청구기호 (이용자가 자료 찾는 번호, 시리즈·COPY 번호 포함)
  authorCode     String?    // 저자기호 (MARC 090 ▼b)
  volume         String?    // 권/호 (예: v.1, 시리즈물일 때 몇 권째인지)
  copyNumber     String?    // 복본 (예: c.1, 같은 자료 중 몇 번째 것인지)
  specialCode    String?    // 별치기호
  shelfNo        String?    // 서가번호
  location       String?    // 소장처 (예: 2층 문학)
  memo           String?    // 메모
  status         String     @default("대출가능")

  createdById    Int?       // 이 실물을 등록한 사서
  createdBy      User?      @relation(fields: [createdById], references: [id])

  createdAt      DateTime   @default(now())
  updatedAt      DateTime   @updatedAt

  loans          Loan[]     // ← 추가 (이 실물의 대출 기록)
  reservations   Reservation[] // ← 추가 (이 실물에 걸린 예약 기록)

  @@unique([libraryId, registrationNo])
  @@index([materialId])
}

// 실물 자료의 '상태' / '별치기호' / '소장처' 드롭다운에 들어갈 값들을 관리합니다.
model CopyOption {
  id        Int      @id @default(autoincrement())

  libraryId Int
  library   Library  @relation(fields: [libraryId], references: [id])

  category  String   // "STATUS" | "SPECIAL_CODE" | "LOCATION" | "FLOOR"
  value     String   // 실제로 화면에 보이고 저장되는 값 (예: "이용가능", "아동", "2층 문학")
  floor     String?  // 소장처(LOCATION)일 때만: 이 소장처가 어느 '층' 값으로 만들어졌는지
  detail    String?  // 소장처(LOCATION)일 때만: 층을 뺀 나머지 세부 위치
  order     Int      @default(0)

  canLoan    Boolean @default(false) // ← 추가: 상태(STATUS)일 때만 의미 있음 — 이 상태의 자료를 대출할 수 있는지
  canReserve Boolean @default(false) // ← 추가: 상태(STATUS)일 때만 의미 있음 — 이 상태의 자료를 예약할 수 있는지

  createdAt DateTime @default(now())

  @@unique([libraryId, category, value])
}

model KormarcTag {
  id            Int      @id @default(autoincrement())
  libraryId     Int
  tag           String
  fieldName     String
  indicators    String?
  subfieldCodes String?
  example       String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  @@unique([libraryId, tag])
}

// 자료 종류(도서, DVD, 기사...)입니다. 관리자가 직접 추가·수정·삭제할 수 있습니다.
model MaterialType {
  id             Int      @id @default(autoincrement())

  libraryId      Int
  library        Library  @relation(fields: [libraryId], references: [id])

  code           String   // 내부적으로 쓰는 코드 (예: "book", "thesis_physical")
  nameKo         String
  nameEn         String
  category       String   // "PHYSICAL"(실물 자료) | "DIGITAL"(디지털 자료)
  usesMarc       Boolean  @default(false) // MARC 편집기를 쓸지, 간단한 입력폼을 쓸지
  order          Int      @default(0)

  maxLoanCount   Int?     // 실물 자료일 때만 사용: 대출 가능 권수
  loanPeriodDays Int?     // 실물 자료일 때만 사용: 대출 일수
  maxReservationCount Int? // 실물 자료일 때만 사용: 예약 가능 권수

  createdAt      DateTime @default(now())

  kdcRules       BookKdcRule[]

  @@unique([libraryId, code])
}

// '도서' 같은 특정 자료 종류 안에서, KDC 분류기호별로 더 세분화된 대출 규칙을 정합니다.
model BookKdcRule {
  id             Int          @id @default(autoincrement())

  libraryId      Int
  library        Library      @relation(fields: [libraryId], references: [id])

  materialTypeId Int
  materialType   MaterialType @relation(fields: [materialTypeId], references: [id])

  kdcPrefix      String       // 예: "657" → 657, 657.1, 657.9 등 포함
  label          String       // 예: "만화"
  maxLoanCount   Int          // 이 KDC의 대출 가능 권수 (상위 자료 종류의 권수를 넘을 수 없음)

  createdAt      DateTime     @default(now())

  @@unique([libraryId, materialTypeId, kdcPrefix])
}

// 회원 구분 (예: 개인회원, 단체회원, 자료지원회원)
model MemberType {
  id        Int      @id @default(autoincrement())

  libraryId Int
  library   Library  @relation(fields: [libraryId], references: [id])

  name      String
  order     Int      @default(0)

  createdAt DateTime @default(now())

  users     User[]
  loanSettings LoanSetting[]

  @@unique([libraryId, name])
}

// 회원구분별 기본 대출 설정 (최대 대출 권수, 최대 예약 권수, 최대 대출 제한 일수, 예약자료 보관일)
model LoanSetting {
  id                  Int        @id @default(autoincrement())

  libraryId           Int
  library             Library    @relation(fields: [libraryId], references: [id])

  memberTypeId        Int
  memberType          MemberType @relation(fields: [memberTypeId], references: [id], onDelete: Cascade)

  maxLoanCount        Int        @default(5)   // 최대 대출 가능 권수
  maxReservationCount Int        @default(3)   // 최대 예약 가능 권수
  maxSuspensionDays   Int?                       // 최대 대출 제한 일수 (비어있으면 상한 없음 = 연체일수만큼 그대로)
  reservationHoldDays Int        @default(3)   // 예약자료 보관일

  createdAt           DateTime   @default(now())

  @@unique([libraryId, memberTypeId])
}

// 대출 기록 (누가, 어떤 실물을, 언제 빌려서 언제까지 반납해야 하는지)
model Loan {
  id          Int       @id @default(autoincrement())

  libraryId   Int
  library     Library   @relation(fields: [libraryId], references: [id])

  copyId      Int
  copy        Copy      @relation(fields: [copyId], references: [id])

  userId      Int
  user        User      @relation(fields: [userId], references: [id])

  loanDate    DateTime  @default(now())  // 대출한 날
  dueDate     DateTime                    // 반납 예정일
  returnedAt  DateTime?                   // 실제로 반납한 날짜/시각 (반납 전이면 비어있음)
  renewCount  Int       @default(0)       // 연장 횟수 (지금은 항상 0, 나중에 연장 기능에서 올라갑니다)

  createdAt   DateTime  @default(now())

  @@index([libraryId])
  @@index([copyId])
  @@index([userId])
}

// 대출 제한 내역 (연체 등으로 회원의 대출이 일정 기간 제한될 때 기록합니다)
model LoanRestriction {
  id          Int      @id @default(autoincrement())

  libraryId   Int
  library     Library  @relation(fields: [libraryId], references: [id])

  userId      Int
  user        User     @relation(fields: [userId], references: [id])

  startDate   DateTime @default(now())  // 대출 제한 시작일
  endDate     DateTime                   // 대출 제한 마지막일 (이 날짜까지는 제한, 다음날부터 자동 해제)
  reason      String?                    // 제한 사유 (예: "연체 5일")

  createdAt   DateTime @default(now())

  @@index([libraryId])
  @@index([userId])
}

// 예약 기록 (누가, 어떤 실물을 예약했는지. 같은 자료+같은 권/호를 가진 실물들이 하나의 그룹으로
// 예약 대기열을 함께 씁니다. status는 RESERVED(예약중) → CANCELED(취소) 또는 FULFILLED(대출완료)로 바뀝니다.)
model Reservation {
  id          Int       @id @default(autoincrement())

  libraryId   Int
  library     Library   @relation(fields: [libraryId], references: [id])

  userId      Int
  user        User      @relation(fields: [userId], references: [id])

  copyId      Int
  copy        Copy      @relation(fields: [copyId], references: [id])

  status      String    @default("RESERVED")   // RESERVED(예약중) | CANCELED(취소) | FULFILLED(대출완료)

  reservedAt  DateTime  @default(now())         // 예약을 건 날짜/시각 (대기 순서 계산 기준)
  holdDueDate DateTime?                          // 보관마감일. 복본을 실제로 보관 중일 때만 값이 있음

  createdAt   DateTime  @default(now())

  @@index([libraryId])
  @@index([userId])
  @@index([copyId])
}

// 게시판 종류 (예: 공지, 열린 게시판, 자료를 신청합니다 등). 도서관마다 자기 게시판 목록을 가집니다.
// 지금은 10개를 고정으로 만들지만, 나중에 '게시판 관리' 기능으로 관리자가 직접 추가/삭제할 수 있도록
// enum이 아니라 테이블로 만들었습니다.
model Board {
  id                Int      @id @default(autoincrement())

  libraryId         Int
  library           Library  @relation(fields: [libraryId], references: [id])

  code              String   // 내부적으로 구분하는 코드 (예: "notice", "openBoard", "materialRequest")
  name              String   // 화면에 보이는 게시판 이름 (예: "공지")
  listStyle         String   @default("LIST")   // "LIST"(제목 목록형) | "THUMBNAIL"(썸네일형)
  order             Int      @default(0)

  allowMemberWrite  Boolean  @default(false)  // 회원이 글을 쓸 수 있는 게시판인지 (열린 게시판, 자료를 신청합니다만 true)
  allowGuestWrite   Boolean  @default(false)  // allowMemberWrite가 true일 때만 의미 있음: 비회원(비로그인)도 글을 쓸 수 있는지
  allowGuestComment Boolean  @default(false)  // 이 게시판의 댓글을 비회원(비로그인)도 쓸 수 있는지

  isMaterialRequest Boolean  @default(false)  // "자료를 신청합니다" 게시판인지 (추가 입력 필드 + 처리 상태 기능이 붙습니다)

  createdAt         DateTime @default(now())

  posts             Post[]

  @@unique([libraryId, code])
}

// 게시글
model Post {
  id                Int       @id @default(autoincrement())

  libraryId         Int
  library           Library   @relation(fields: [libraryId], references: [id])

  boardId           Int
  board             Board     @relation(fields: [boardId], references: [id])

  title             String
  content           String    // 리치 텍스트 에디터로 작성한 본문 (HTML 형태로 저장)
  thumbnailUrl      String?   // 썸네일형 게시판에서 목록에 보여줄 대표 이미지

  authorUserId      Int?                             // 회원(또는 관리자)이 쓴 글이면 채워짐
  authorUser        User?     @relation(fields: [authorUserId], references: [id])
  guestName         String?                           // 비회원이 쓴 글이면 채워짐 (작성자 이름)
  guestPasswordHash String?                           // 비회원이 쓴 글이면 채워짐 (수정/삭제용 비밀번호, 암호화 저장)

  viewCount         Int       @default(0)

  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt

  comments          Comment[]
  materialRequest   MaterialRequest?

  @@index([libraryId])
  @@index([boardId])
}

// 댓글
model Comment {
  id                Int       @id @default(autoincrement())

  libraryId         Int
  library           Library   @relation(fields: [libraryId], references: [id])

  postId            Int
  post              Post      @relation(fields: [postId], references: [id])

  content           String

  authorUserId      Int?
  authorUser        User?     @relation(fields: [authorUserId], references: [id])
  guestName         String?
  guestPasswordHash String?

  createdAt         DateTime  @default(now())

  @@index([libraryId])
  @@index([postId])
}

// "자료를 신청합니다" 게시글에 붙는 추가 정보 (신청 자료 종류, 저자, 처리 상태)
model MaterialRequest {
  id          Int      @id @default(autoincrement())

  postId      Int      @unique
  post        Post     @relation(fields: [postId], references: [id])

  requestType String   // 신청 자료 종류 (예: "도서", "DVD", "공구")
  author      String?  // 저자

  status      String   @default("REQUESTED")  // REQUESTED(신청) | PURCHASING(구입 진행 중) | PURCHASED(구입 완료) | NOT_PURCHASED(미구입)

  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}
```

저장합니다.
